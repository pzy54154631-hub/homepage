\section*{Selected Research Projects}
\researchentry{Ambiguity Is Not One Scalar: Explicit Referents Improve VQA Routing}{2026}{Conditional multimodal inference; selective prediction; calibration}
\begin{itemize}
    \item Designed a two-stage VQA router over \textbf{ANSWER / CLARIFY / ABSTAIN}, using explicit referent enumeration for uncertain or focus-sensitive inputs.
    \item On a 26,425-example diagnostic benchmark, referent enumeration recovered \textbf{20/22} low-score focus-ambiguity false negatives, versus \textbf{0/22} for answer self-consistency.
\end{itemize}

\researchentry{PW-OFPC: Precision-Weighted Object-File Predictive Coding}{2025--2026}{Uncertainty-aware inference; object-centric reasoning; candidate comparison}
\begin{itemize}
    \item Built an uncertainty-aware abstract visual reasoning pipeline that preserves object-file posteriors and compares answer candidates jointly using predictive-coding residuals.
    \item On public RAVEN, the inference ladder improved macro accuracy from \textbf{0.537 to 0.857}; controlled experiments separated object-formation failures from residual and candidate-selection failures.
\end{itemize}
